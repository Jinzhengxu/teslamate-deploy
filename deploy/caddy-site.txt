# TeslaMate —— 本段由 deploy/deploy.sh 按上下两行标记【整块】幂等替换，
# 请不要在 BEGIN / END 之间放别的站点配置，否则重新部署时会被覆盖掉。
# 下面用双下划线包起来的大写单词都是占位符（域名、认证指令名、用户名、密码哈希、上游），写入时替换。
__DOMAIN__ {
	encode zstd gzip

	# Grafana 自带登录。它开了 SERVE_FROM_SUB_PATH，要收到带 /grafana 前缀的原始路径，
	# 所以这里用 handle 而不是 handle_path。
	@grafana path /grafana /grafana/*
	handle @grafana {
		reverse_proxy teslamate-grafana:3000
	}

	# TeslaMate 本身没有任何登录：谁打开都能看到车的实时位置、改设置，
	# 所以这一层密码是必须的，不是可选项。
	handle {
		__BASIC_AUTH__ {
			__AUTH_USER__ __AUTH_HASH__
		}
		# 开了换肤时上游是「teslamate-theme:8080 teslamate:4000」：平时走换肤代理，
		# 它没起来或连不上时，当次请求改走 teslamate 本体（原版界面），10 秒后再试换肤代理。
		# 只有一个上游时 fail_duration 写成 0（关掉被动健康检查），免得 TeslaMate 重启后多等一阵 503
		reverse_proxy __UPSTREAMS__ {
			lb_policy first
			lb_try_duration 5s
			fail_duration __FAIL_DURATION__
		}
	}

	header {
		X-Content-Type-Options nosniff
	}
}
# <<< teslamate END
