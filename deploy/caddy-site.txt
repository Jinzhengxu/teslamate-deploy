	# TeslaMate 本身没有任何登录：谁打开都能看到车的实时位置、改设置，所以这一层密码是必须的。
	# 密码只在登录页验一次，验过就发一个长期 cookie（tm_auth），之后这台设备不用再输。
	# cookie 的值是 .env 里的 TM_WEB_TOKEN，重置密码时会换掉，所有设备随之退出。
	# 验证全在 Caddy 里做，不依赖换肤代理：换肤代理挂了不会绕过登录。

	# 登录页的「登录」按钮用 fetch 把用户名密码（Basic 认证头）发到这里
	handle /_auth/verify {
		route {
			__BASIC_AUTH__ {
				__AUTH_USER__ __AUTH_HASH__
			}
			header Set-Cookie "tm_auth=__AUTH_TOKEN__; Path=/; Max-Age=34560000; Secure; HttpOnly; SameSite=Lax"
			header Cache-Control no-store
			respond 204
		}
	}

	# 只退出当前这台设备；要让所有设备都退出，就重置密码
	handle /_auth/logout {
		header Set-Cookie "tm_auth=; Path=/; Max-Age=0; Secure; HttpOnly; SameSite=Lax"
		redir * / 302
	}

	# 用单引号：老版本 Caddy（2.4）解析这一行时会把双引号吃掉，CEL 就不认得这是字符串了
	@authed expression {http.request.cookie.tm_auth} == '__AUTH_TOKEN__'
	handle @authed {
		# 每次打开页面顺手把 cookie 续到 400 天（浏览器允许的上限），常用的设备就一直不用重新登录
		@page header Sec-Fetch-Mode navigate
		header @page +Set-Cookie "tm_auth=__AUTH_TOKEN__; Path=/; Max-Age=34560000; Secure; HttpOnly; SameSite=Lax"

		# Grafana 开了 auth proxy：认 X-WEBAUTH-USER 头里的用户名，不再单独要密码。
		# header_up 是覆盖，浏览器自己带来的同名头到不了 Grafana。
		# 它开了 SERVE_FROM_SUB_PATH，要收到带 /grafana 前缀的原始路径，所以用 handle 而不是 handle_path。
		@grafana path /grafana /grafana/*
		handle @grafana {
			reverse_proxy teslamate-grafana:3000 {
				header_up X-WEBAUTH-USER admin
			}
		}

		# 开了换肤时上游是「teslamate-theme:8080 teslamate:4000」：平时走换肤代理，
		# 它没起来或连不上时，当次请求改走 teslamate 本体（原版界面），10 秒后再试换肤代理。
		# 只有一个上游时 fail_duration 写成 0（关掉被动健康检查），免得 TeslaMate 重启后多等一阵 503
		handle {
			reverse_proxy __UPSTREAMS__ {
				lb_policy first
				lb_try_duration 5s
				fail_duration __FAIL_DURATION__
			}
		}
	}
